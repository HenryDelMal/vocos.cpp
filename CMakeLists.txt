cmake_minimum_required(VERSION 3.23)
project(vocos_native LANGUAGES CXX)
set(VOCOS_EIGEN_DIR "" CACHE PATH "Directory containing Eigen headers")
if(NOT EXISTS "${VOCOS_EIGEN_DIR}/Eigen/Core")
  message(FATAL_ERROR "Set VOCOS_EIGEN_DIR to an Eigen source/include directory")
endif()
add_library(vocos src/vocos.cpp)
target_compile_features(vocos PUBLIC cxx_std_20)
target_include_directories(vocos PUBLIC include PRIVATE "${VOCOS_EIGEN_DIR}")
if(MSVC)
  target_compile_options(vocos PRIVATE /W4)
else()
  target_compile_options(vocos PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(vocos_fixture tools/fixture.cpp)
target_link_libraries(vocos_fixture PRIVATE vocos)
